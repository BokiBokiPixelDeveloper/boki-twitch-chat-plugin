find_package(Qt6 6.9 REQUIRED COMPONENTS Test)
add_executable(chat-tests chat-tests.cpp chat-tests.hpp ${CHAT_EMOJI_RESOURCE})
set_target_properties(chat-tests PROPERTIES AUTOMOC ON AUTORCC ON)
target_link_libraries(chat-tests PRIVATE chat-support Qt6::Test)
target_compile_definitions(chat-tests PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME chat-tests COMMAND chat-tests -v1 -o -,txt
    -o "${CMAKE_CURRENT_BINARY_DIR}/chat-tests.txt,txt")
if(WIN32)
    set_tests_properties(chat-tests PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=windows;QT_PLUGIN_PATH=$<TARGET_FILE_DIR:Qt6::Core>/../plugins;QT_FORCE_STDERR_LOGGING=1;QT_LOGGING_RULES=bokis.render.emoji.debug=true" TIMEOUT 30)
else()
    set_tests_properties(chat-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()

add_executable(updater-tests updater-tests.cpp ../src/updater/update-checker.cpp)
set_target_properties(updater-tests PROPERTIES AUTOMOC ON)
target_include_directories(updater-tests PRIVATE ../src $<TARGET_PROPERTY:OBS::libobs,INTERFACE_INCLUDE_DIRECTORIES>)
target_link_libraries(updater-tests PRIVATE post-exit-support Qt6::Core Qt6::Network Qt6::Test)
target_compile_definitions(updater-tests PRIVATE QT_NO_KEYWORDS)
add_test(NAME updater-tests COMMAND updater-tests)
set_tests_properties(updater-tests PROPERTIES TIMEOUT 30)

if(WIN32)
    add_library(updater-test-image SHARED windows-test-image.cpp)
    add_executable(post-exit-tests post-exit-windows-tests.cpp)
    target_compile_definitions(post-exit-tests PRIVATE TEST_IMAGE_PATH="$<TARGET_FILE:updater-test-image>")
    add_dependencies(post-exit-tests updater-test-image)
else()
    add_executable(post-exit-tests post-exit-tests.cpp)
endif()
set_target_properties(post-exit-tests PROPERTIES AUTOMOC ON)
target_link_libraries(post-exit-tests PRIVATE post-exit-support Qt6::Test)
target_compile_definitions(post-exit-tests PRIVATE HELPER_PATH="$<TARGET_FILE:bokis-twitch-chat-updater>")
add_dependencies(post-exit-tests bokis-twitch-chat-updater)
add_test(NAME post-exit-tests COMMAND post-exit-tests)
set_tests_properties(post-exit-tests PROPERTIES TIMEOUT 90)
